@dataclass(frozen=True)
class DailyReportAgentRequest:
    target_date: str
    mode: str = "replay"
    update_decision_ledger: bool = True
    review_prior_participation: bool = True
    output_variant: str = "full"


@dataclass(frozen=True)
class DailyReportAgentDependencies:
    load_data: Callable[[DailyReportAgentRequest], dict[str, Any]] | None = None
    prepare_market_language_translation: Callable[..., dict[str, Any]] = (
        prepare_market_language_translation
    )
    build_report_outputs: Callable[..., dict[str, str]] = build_report_outputs
    save_report_outputs: Callable[..., dict[str, str]] = save_report_outputs
    write_replay_market_language_translation: Callable[..., dict[str, Any]] = (
        write_replay_market_language_translation
    )
    write_decision_ledger_from_markdown: Callable[..., dict[str, Any]] = (
        write_decision_ledger_artifacts_from_markdown_report
    )
    render_position_strategy_report: Callable[..., str] | None = None
    audit_run: Callable[..., dict[str, Any]] = audit_daily_report_run
    build_observer: Callable[..., Any] = build_report_run_observer
    now: Callable[[], datetime] = datetime.now


def _default_load_data(request: DailyReportAgentRequest) -> dict[str, Any]:
    if request.mode == "replay":
        replay_result = replay_from_snapshot(request.target_date)
        if isinstance(replay_result, tuple):
            return replay_result[0] or {}
        return replay_result or {}
    raise UnsupportedRunModeError("当前 runner 仅支持 replay 模式")


class RunnerFailure(RuntimeError):
    code = "runner_error"


class ReplayDataMissingError(RunnerFailure):
    code = "replay_data_missing"


class UnsupportedRunModeError(RunnerFailure):
    code = "unsupported_run_mode"


class ReportAcceptanceError(RunnerFailure):
    code = "report_acceptance_failed"

    def __init__(self, audit_result: dict[str, Any]):
        self.audit_result = audit_result
        details = []
        for finding in audit_result.get("findings") or []:
            code = str(finding.get("code") or self.code).strip()
            message = str(finding.get("message") or "").strip()
            details.append(f"{code}: {message}" if message else code)
        super().__init__("；".join(details) or "日报检查未通过，停止生成")


class DailyReportAgentRunner:
    def __init__(
        self,
        *,
        root_dir: str | Path = ".",
        dependencies: DailyReportAgentDependencies | None = None,
    ):
        self.root_dir = Path(root_dir)
        self._injected_legacy_flow = dependencies is not None
        self.dependencies = dependencies or DailyReportAgentDependencies()

    def _run_current_sector_reading(self, request: DailyReportAgentRequest) -> dict[str, Any]:
        """Route the production runner to the current reading entry.

        Dependency-injected legacy orchestration remains available to its old
        tests and historical tooling, but the normal agent runner must not
        rebuild account decisions or the retired full report.
        """
        from daily_report_entry_demo import main as reading_main

        previous_argv = sys.argv[:]
        sys.argv = [
            "daily_report_entry_demo.py",
            request.target_date,
            "--replay",
            "--snapshot-source",
            "replay" if request.mode == "replay" and request.output_variant == "sector-replay" else "official",
        ]
        try:
            exit_code = reading_main()
        finally:
            sys.argv = previous_argv
        return {
            "status": "success" if exit_code == 0 else "error",
            "mode": "sector",
            "run_dir": str(self.root_dir / "reports" / "runs" / request.target_date),
            "error": None if exit_code == 0 else {"message": "板块阅读入口未生成"},
        }

    def run(self, request: DailyReportAgentRequest) -> dict[str, Any]:
        if not self._injected_legacy_flow:
            return self._run_current_sector_reading(request)
        started_at = self.dependencies.now().isoformat(timespec="seconds")
        run_dir = self.root_dir / "reports" / "runs" / request.target_date
        run_dir.mkdir(parents=True, exist_ok=True)

        status = "success"
        error: dict[str, str] | None = None
        data: dict[str, Any] = {}
        report_outputs: dict[str, str] = {}
        output_paths: dict[str, str] = {}
        decision_ledger_result: dict[str, Any] | None = None
        audit_result: dict[str, Any] | None = None
        market_language_translation: dict[str, Any] = {}
        market_language_storage: dict[str, Any] = {}
        observer = self.dependencies.build_observer(
            run_name=f"daily-report.agent.{request.target_date}",
            trace_input={
                "target_date": request.target_date,
                "mode": request.mode,
                "update_decision_ledger": request.update_decision_ledger,
                "review_prior_participation": request.review_prior_participation,
            },
            metadata={"runner": "agent_runner.daily_report_agent_runner"},
        )

        try:
            if request.mode != "replay":
                raise UnsupportedRunModeError("当前 runner 仅支持 replay 模式")
            load_data = self.dependencies.load_data or _default_load_data
            data = load_data(request) or {}
            observer.record_step(
                "load_data",
                status="success",
                input_payload={"target_date": request.target_date, "mode": request.mode},
                output_payload={"data_keys": list(data.keys())[:30], "empty": not bool(data)},
            )
            if request.mode == "replay" and not data:
                raise ReplayDataMissingError(f"回放数据不存在，停止生成：{request.target_date}")
            normalize_inferred_previous_target_records(data)
            market_language_translation = self.dependencies.prepare_market_language_translation(
                data,
                target_date=request.target_date,
                analysis_mode=request.mode,
                root_dir=self.root_dir,
            )
            report_outputs = self.dependencies.build_report_outputs(
                data,
                request.target_date,
                run_mode=request.mode,
                emit_preflight=False,
                persist_runtime_artifacts=False,
                market_language_translation=market_language_translation,
            )
            if existing_position_summary_from_context(data):
                render_position_strategy = self.dependencies.render_position_strategy_report
                if render_position_strategy is None:
                    from portfolio.position_strategy_report import render_position_strategy_report

                    render_position_strategy = render_position_strategy_report
                position_report_text = render_position_strategy(
                    data,
                    request.target_date,
                    [],
                    source_report_markdown=_select_full_report_markdown(report_outputs, ""),
                    output_root=run_dir,
                    prefer_postgres_history=True,
                )
                position_timeline = extract_position_timeline_markdown(position_report_text)
                if position_timeline:
                    report_outputs = dict(report_outputs)
                    full_key = "full" if "full" in report_outputs else next(iter(report_outputs), "full")
                    report_outputs[full_key] = prepend_position_timeline_markdown(
                        report_outputs.get(full_key, ""),
                        position_timeline,
                    )
            observer.record_step(
                "build_report_outputs",
                status="success",
                input_payload={"target_date": request.target_date, "mode": request.mode},
                output_payload={"output_variants": list(report_outputs.keys())},
            )
            assert_report_required_facts_complete(data, request.target_date, run_mode=request.mode)
            observer.record_step(
                "preflight_check",
                status="success",
                input_payload={"target_date": request.target_date, "mode": request.mode},
                output_payload={"result": "facts_complete"},
            )
            full_markdown = _select_full_report_markdown(report_outputs, "")
            pre_save_audit = self.dependencies.audit_run(
                target_date=request.target_date,
                data=data,
                report_markdown=full_markdown,
                source_path=_planned_full_report_path(run_dir, request.target_date, request.mode),
                prefer_postgres_history=True,
                decision_ledger_result=None,
                decision_ledger_required=False,
            )
            observer.record_step(
                "pre_save_audit",
                status="success" if pre_save_audit.get("passed") else "error",
                input_payload={"target_date": request.target_date},
                output_payload=_compact_audit(pre_save_audit),
            )
            if not pre_save_audit.get("passed"):
                raise ReportAcceptanceError(pre_save_audit)
            output_paths = self.dependencies.save_report_outputs(
                report_outputs,
                request.target_date,
                mode=request.mode,
                output_dir=str(run_dir),
            )
            observer.record_step(
                "save_report_outputs",
                status="success",
                input_payload={"target_date": request.target_date, "mode": request.mode},
                output_payload={"output_variants": list(output_paths.keys())},
            )
            full_report_path = _select_full_report_path(output_paths)
            saved_full_markdown = read_verified_saved_report_markdown(
                full_report_path,
                full_markdown,
            )

            if request.update_decision_ledger and full_report_path:
                decision_ledger_result = self.dependencies.write_decision_ledger_from_markdown(
                    full_report_path,
                    trade_date=request.target_date,
                    market_branch="以最终日报为准",
                    sector_relationship_state="以最终日报为准",
                    root_dir=str(self.root_dir),
                    generated_at=started_at,
                )
                observer.record_step(
                    "decision_ledger",
                    status="success",
                    input_payload={"target_date": request.target_date},
                    output_payload=_compact_decision_ledger(decision_ledger_result),
                )

            audit_result = self.dependencies.audit_run(
                target_date=request.target_date,
                data=data,
                report_markdown=saved_full_markdown,
                source_path=full_report_path,
                prefer_postgres_history=True,
                decision_ledger_result=decision_ledger_result,
                decision_ledger_required=request.update_decision_ledger,
            )
            observer.record_step(
                "audit_run",
                status="success" if audit_result.get("passed") else "needs_review",
                input_payload={"target_date": request.target_date},
                output_payload=_compact_audit(audit_result),
            )
            if not audit_result.get("passed"):
                status = "needs_review"
            elif request.mode == "replay":
                market_language_storage = (
                    self.dependencies.write_replay_market_language_translation(
                        market_language_translation,
                        root_dir=self.root_dir,
                    )
                    or {}
                )
        except Exception as exc:
            status = "error"
            error = {"type": type(exc).__name__, "message": str(exc)}
            observer.record_step(
                "agent_run",
                status="error",
                input_payload={"target_date": request.target_date, "mode": request.mode},
                output_payload={},
                error=exc,
            )
            if isinstance(exc, ReportAcceptanceError):
                audit_result = exc.audit_result
            else:
                finding_code = exc.code if isinstance(exc, RunnerFailure) else "runner_error"
                audit_result = {
                    "target_date": request.target_date,
                    "passed": False,
                    "usable": False,
                    "findings": [
                        {
                            "source": "runner",
                            "code": finding_code,
                            "message": str(exc),
                        }
                    ],
                    "finding_count": 1,
                }

        finished_at = self.dependencies.now().isoformat(timespec="seconds")
        payload = {
            "target_date": request.target_date,
            "mode": request.mode,
            "status": status,
            "started_at": started_at,
            "finished_at": finished_at,
            "run_dir": str(run_dir),
            "outputs": output_paths,
            "audit": audit_result,
            "decision_ledger": _compact_decision_ledger(decision_ledger_result),
            "market_language_translation": market_language_storage,
            "error": error,
        }
        report_quality_score = build_report_quality_score(
            target_date=request.target_date,
            run_status=status,
            mode=request.mode,
            audit_result=audit_result,
            decision_ledger_result=decision_ledger_result,
            output_paths=output_paths,
            error=(error or {}).get("message") if error else None,
            report_expected=True,
            decision_ledger_expected=request.update_decision_ledger,
        )
        report_quality_ledger = persist_report_quality_score(
            report_quality_score,
            root_dir=self.root_dir,
            run_id=f"daily-report.agent.{request.target_date}",
            recorded_at=finished_at,
        )
        report_quality_score["local_ledger"] = report_quality_ledger
        payload["report_quality_score"] = report_quality_score
        observer.finish(
            status=status,
            output_payload={
                "target_date": request.target_date,
                "mode": request.mode,
                "output_variants": list(output_paths.keys()),
                "audit_passed": bool((audit_result or {}).get("passed")),
                "finding_count": int((audit_result or {}).get("finding_count") or 0),
                "report_quality_score": report_quality_score,
                "report_quality_ledger": report_quality_ledger,
            },
        )

        self._write_json(run_dir / "run.json", payload)
        self._write_json(run_dir / "audit.json", audit_result)
        (run_dir / "overview.md").write_text(_render_overview(payload), encoding="utf-8")
        (run_dir / "review_queue.md").write_text(
            _render_review_queue(audit_result, decision_ledger_result),
            encoding="utf-8",
        )
        return payload

    @staticmethod
    def _write_json(path: Path, payload: Any) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(
            json.dumps(payload, ensure_ascii=False, indent=2, sort_keys=True),
            encoding="utf-8",
        )


def _select_full_report_path(output_paths: dict[str, str]) -> str:
    if "full" in output_paths:
        return output_paths["full"]
    if output_paths:
        return next(iter(output_paths.values()))
    return ""


def _planned_full_report_path(run_dir: Path, target_date: str, mode: str) -> str:
    suffix = "_replay" if mode == "replay" else ""
    return str(run_dir / f"{target_date}_A股日报_完整版{suffix}.md")


def _select_full_report_markdown(report_outputs: dict[str, str], full_report_path: str) -> str:
    if "full" in report_outputs:
        return report_outputs["full"]
    if report_outputs:
        return next(iter(report_outputs.values()))
    if full_report_path:
        return Path(full_report_path).read_text(encoding="utf-8")
    return ""


def _compact_decision_ledger(result: dict[str, Any] | None) -> dict[str, Any]:
    if not result:
        return {"case_count": 0}
    return {
        "case_count": int(result.get("case_count") or 0),
        "snapshot_path": result.get("snapshot_path", ""),
        "index_path": result.get("index_path", ""),
        "fact_db_sync": result.get("fact_db_sync") or {},
    }


def _compact_audit(result: dict[str, Any] | None) -> dict[str, Any]:
    result = result or {}
    return {
        "passed": bool(result.get("passed")),
        "usable": bool(result.get("usable")),
        "finding_count": int(result.get("finding_count") or 0),
        "finding_codes": [
            finding.get("code", "")
            for finding in list(result.get("findings") or [])[:20]
            if isinstance(finding, dict)
        ],
    }


def _render_overview(payload: dict[str, Any]) -> str:
    audit = payload.get("audit") or {}
    ledger = payload.get("decision_ledger") or {}
    lines = [
        f"# Daily Report Agent Run {payload.get('target_date', '')}",
        "",
        f"- status: {payload.get('status', '')}",
        f"- mode: {payload.get('mode', '')}",
        f"- usable: {audit.get('usable', False)}",
        f"- audit_passed: {audit.get('passed', False)}",
        f"- finding_count: {audit.get('finding_count', 0)}",
        f"- decision_ledger_cases: {ledger.get('case_count', 0)}",
        "",
        "## Outputs",
    ]
    for variant, path in (payload.get("outputs") or {}).items():
        lines.append(f"- {variant}: {path}")
    if payload.get("error"):
        lines.extend(
            [
                "",
                "## Error",
                f"- {payload['error'].get('type')}: {payload['error'].get('message')}",
            ]
        )
    return "\n".join(lines).rstrip() + "\n"


def _render_review_queue(
    audit: dict[str, Any] | None,
    decision_ledger_result: dict[str, Any] | None,
) -> str:
    lines = ["# Review Queue", ""]
    findings = (audit or {}).get("findings") or []
    pending_records = _pending_review_records(decision_ledger_result)

    if not findings and not pending_records:
        lines.append("- [ ] No manual review items from this run.")
        return "\n".join(lines).rstrip() + "\n"

    for finding in findings:
        lines.append(f"- [ ] {finding.get('source', 'runner')}:{finding.get('code', 'unknown')}")
        message = finding.get("message") or ""
        if message:
            lines.append(f"  - reason: {message}")
        lines.append("  - human judgment:")
        lines.append("  - follow-up:")

    for record in pending_records:
        stock_name = record.get("stock_name") or record.get("name") or ""
        stock_code = record.get("stock_code") or record.get("code") or ""
        lines.append(f"- [ ] participation_review pending: {stock_name}({stock_code})")
        lines.append("  - reason: prior planned participation is waiting for explicit facts.")
        lines.append("  - human judgment:")
        lines.append("  - follow-up:")

    return "\n".join(lines).rstrip() + "\n"


def _pending_review_records(decision_ledger_result: dict[str, Any] | None) -> list[dict[str, Any]]:
    index = (decision_ledger_result or {}).get("index") or {}
    records: list[dict[str, Any]] = []
    for source_record in index.get("records") or index.get("latest_records") or []:
        if isinstance(source_record, dict):
            records.append(source_record)
    for stock_records in (index.get("by_stock_code") or {}).values():
        if isinstance(stock_records, list):
            records.extend(record for record in stock_records if isinstance(record, dict))
        elif isinstance(stock_records, dict):
            records.append(stock_records)
    pending = []
    for record in records:
        status = record.get("participation_review_status")
        if status in {
            None,
            "",
            "pending",
            "waiting",
            "waiting_for_facts",
            "等待复盘",
            "等待窗口复盘",
        }:
            pending.append(record)
    return pending
