def _run_daily_report_consensus_short(
    data,
    target_date,
    *,
    run_mode,
    output_root=None,
    reviewer=None,
    model_version="",
    database_service=None,
    run_id=None,
    now_fn=None,
):
    if not isinstance(data, dict) or not data:
        raise ReportCriticalDataMissingError(
            "日报事实为空，关键数据不完整，停止生成"
        )
    now_fn = now_fn or _daily_report_consensus_now
    frozen_at = now_fn()
    fact_rows = _short_consensus_fact_rows(data, target_date)
    source_facts = data.get("candidate_decision_evidence_facts") or {}
    decision_cutoff_at = str(
        source_facts.get("decision_cutoff_at")
        or data.get("decision_cutoff_at")
        or ""
    ).strip()
    if not decision_cutoff_at and run_mode == LEDGER_MODE_REPLAY:
        raise ReportCriticalDataMissingError(
            "历史事实缺少冻结截止时间，关键数据不完整，停止生成"
        )
    decision_cutoff_at = decision_cutoff_at or frozen_at

    formal_input = prepare_formal_decision_input(data, report_date=target_date)
    if not isinstance(formal_input, dict) or formal_input.get("source_valid") is not True:
        raise ReportCriticalDataMissingError(
            "正式候选事实未通过日期与完整性核对，关键数据不完整，停止生成"
        )
    workflow = data.get("selection_workflow")
    if (
        not isinstance(workflow, dict)
        or workflow.get("schema_version") != "selection_workflow.v1"
        or str(workflow.get("target_date") or "").strip() != target_date
        or not isinstance(workflow.get("final"), dict)
    ):
        raise ReportCriticalDataMissingError(
            "唯一候选提案尚未形成，关键数据不完整，停止生成"
        )
    proposed_workflow = deepcopy(workflow)
    proposed_workflow["proposed_final"] = deepcopy(workflow["final"])

    llm_service = None
    resolved_reviewer = reviewer
    resolved_model_version = str(model_version or "").strip()
    if resolved_reviewer is None:
        llm_service = get_llm_service()
        resolved_reviewer = make_llm_reviewer(llm_service)
        resolved_model_version = resolved_model_version or str(
            getattr(llm_service, "model", "") or ""
        ).strip()
    if not resolved_model_version:
        raise ReportCriticalDataMissingError(
            "复核模型版本缺失，关键数据不完整，停止生成"
        )
    reviewer_model_version = str(
        getattr(resolved_reviewer, "model_version", "") or ""
    ).strip()
    if reviewer_model_version and reviewer_model_version != resolved_model_version:
        raise ReportCriticalDataMissingError(
            "实际复核模型与冻结批次不一致，停止生成"
        )
    review_seed_supported = bool(
        getattr(resolved_reviewer, "seed_supported", False)
    )

    batch = build_frozen_batch(
        target_date=target_date,
        query_date=source_facts.get("query_date"),
        formal_input=formal_input,
        selection_workflow=proposed_workflow,
        facts=fact_rows,
        required_fact_ids=("candidate_decision_evidence_facts",),
        decision_cutoff_at=decision_cutoff_at,
        collected_at=source_facts.get("collected_at") or frozen_at,
        run_id=run_id,
        model_version=resolved_model_version,
        seed_supported=review_seed_supported,
        knowledge_versions=data.get("knowledge_versions") or {},
        review_context={
            "candidate_selection": deepcopy(
                formal_input.get("candidate_selection") or {}
            ),
            "candidate_ledger": deepcopy(formal_input.get("candidate_ledger") or []),
            "candidate_observations": deepcopy(
                formal_input.get("candidate_observations") or []
            ),
            "account_qualifications": deepcopy(
                formal_input.get("account_qualifications") or {}
            ),
        },
    )
    if batch.get("input_status") != "ready":
        raise ReportCriticalDataMissingError(
            "关键数据不完整，停止生成正式决定：{}".format(
                "、".join(batch.get("input_issue_codes") or ["事实核对未通过"])
            )
        )

    storage_service = database_service
    if run_mode != LEDGER_MODE_REPLAY:
        storage_service = storage_service or _postgres_daily_report_service()
        checker = getattr(storage_service, "is_postgres_backed", None)
        if not callable(checker) or not checker():
            raise ReportCriticalDataMissingError(
                "PostgreSQL 共识记录库不可用，停止生成"
            )

    review_started_at = now_fn()
    reviews = run_consensus_reviews(batch, resolved_reviewer)
    consensus = calculate_consensus(batch, reviews)
    committed_workflow = apply_consensus_to_selection_workflow(
        proposed_workflow,
        consensus,
    )
    formal_account_decision = build_formal_account_decision(
        formal_input,
        selection_workflow=committed_workflow,
        report_date=target_date,
    )
    decision_hash = str(formal_account_decision.get("decision_hash") or "").strip()
    if not decision_hash:
        raise ReportCriticalDataMissingError(
            "正式账户决定未形成，关键数据不完整，停止生成"
        )
    if consensus.get("status") == "stable_object" and not formal_account_decision.get(
        "formal_execution_object"
    ):
        raise ReportCriticalDataMissingError(
            "共识提案与正式账户决定不一致，停止生成"
        )

    from report import runtime as report_runtime

    decision_data = dict(data)
    decision_data.pop("decision_product_freeze", None)
    decision_data.pop("decision_product_freeze_review", None)
    decision_data.update(
        {
            "target_date": target_date,
            "selection_workflow": deepcopy(committed_workflow),
            "formal_decision_input": deepcopy(formal_input),
            "formal_account_decision": deepcopy(formal_account_decision),
            "execution_stage_context": {
                "selection_workflow": deepcopy(committed_workflow),
            },
            "market_research_snapshot": {},
        }
    )
    execution_context = decision_data["execution_stage_context"]
    market_snapshot = decision_data["market_research_snapshot"]
    position_decision = _first_dict_from_sources(
        "position_decision",
        data,
        data.get("execution_stage_context"),
        data.get("market_research_snapshot"),
    )
    position_decision = report_runtime._sync_position_decision_with_selection_workflow(
        deepcopy(position_decision),
        committed_workflow,
    )
    position_decision = report_runtime._store_runtime_position_decision(
        position_decision,
        data=decision_data,
        execution_stage_context=execution_context,
        market_research_snapshot=market_snapshot,
    )
    workflow_final = (
        committed_workflow.get("final")
        if isinstance(committed_workflow.get("final"), dict)
        else {}
    )
    selected_candidate = (
        workflow_final.get("execution_object")
        if isinstance(workflow_final.get("execution_object"), dict)
        else {}
    )
    final_target_builder = report_runtime._build_runtime_final_target_resolution
    final_target_kwargs = {
        "data": decision_data,
        "selected_position_candidate": selected_candidate,
        "position_decision": position_decision,
        "market_research_snapshot": market_snapshot,
        "execution_stage_context": execution_context,
    }
    from inspect import signature

    if "formal_account_decision" in signature(final_target_builder).parameters:
        final_target_kwargs["formal_account_decision"] = formal_account_decision
    final_target_resolution = final_target_builder(
        **final_target_kwargs
    )
    final_target_resolution = report_runtime._store_runtime_final_target_resolution(
        final_target_resolution,
        data=decision_data,
        execution_stage_context=execution_context,
        market_research_snapshot=market_snapshot,
        phase="consensus_short",
    )
    two_layer_decision_payload = report_runtime.build_two_layer_decision_payload(
        target_date=target_date,
        final_target_resolution=final_target_resolution,
        position_decision=position_decision,
        execution_stage_context=execution_context,
    )
    report_runtime._store_runtime_two_layer_decision_payload(
        two_layer_decision_payload,
        data=decision_data,
        execution_stage_context=execution_context,
        market_research_snapshot=market_snapshot,
    )
    decision_freeze = build_decision_product_freeze(
        decision_data,
        target_date=target_date,
        artifact_origin=(
            "replay_consensus" if run_mode == LEDGER_MODE_REPLAY else "fresh_consensus"
        ),
    )
    freeze_valid, freeze_errors = validate_decision_product_freeze(
        decision_freeze,
        target_date=target_date,
    )
    if not freeze_valid or decision_freeze.get("availability") != "complete":
        freeze_failures = list(freeze_errors or [])
        if decision_freeze.get("availability") != "complete":
            freeze_failures.extend(decision_freeze.get("missing_fields") or [])
        raise ReportCriticalDataMissingError(
            "正式决定冻结结果未通过核对，停止生成：{}".format(
                "、".join(dict.fromkeys(freeze_failures or ["冻结结果缺失"]))
            )
        )

    decision_at = now_fn()
    for key in (
        "selection_workflow",
        "final_target_resolution",
        "position_decision",
        "two_layer_decision_payload",
        "formal_decision_input",
        "formal_account_decision",
    ):
        data[key] = deepcopy(decision_data[key])
    data["decision_product_freeze"] = deepcopy(decision_freeze)
    settled_trade_dates = formal_input.get("trade_dates") or []
    current_metrics = {
        target_date: {
            "consensus_status": consensus.get("status"),
            "confirmation_count": consensus.get("confirmation_count"),
            "denominator": consensus.get("denominator"),
            "decision_hash": decision_hash,
        }
    }
    periods = build_period_windows(
        target_date=target_date,
        settled_trade_dates=settled_trade_dates,
        metrics_by_date=current_metrics,
    )
    batch.update(
        {
            "decision_hash": decision_hash,
            "freeze_hash": decision_freeze.get("freeze_hash"),
            "decision_at": decision_at,
            "consensus": consensus,
            "periods": periods,
            "decision_freeze": deepcopy(decision_freeze),
        }
    )
    batch.setdefault("time_roles", {})["decision_at"] = decision_at
    funnel_records = build_funnel_records(batch, reviews, consensus)
    funnel = funnel_summary(funnel_records)
    batch["funnel"] = funnel
    summary_text = render_consensus_summary(
        batch,
        reviews,
        consensus,
        periods=periods,
        funnel=funnel,
    )
    run_records = [
        {
            "schema_version": "daily_report_consensus_run.v1",
            "run_id": batch.get("run_id"),
            "target_date": target_date,
            "run_mode": run_mode,
            "status": "success",
            "started_at": review_started_at,
            "ended_at": decision_at,
            "input_hash": batch.get("input_hash"),
            "proposal_hash": batch.get("proposal_hash"),
            "decision_hash": decision_hash,
            "freeze_hash": decision_freeze.get("freeze_hash"),
            "review_timing": review_timing_summary(reviews),
        }
    ]
    root_dir = Path(output_root) if output_root is not None else Path(__file__).resolve().parents[1]
    output_dir = consensus_output_directory(
        root_dir,
        target_date=target_date,
        run_id=batch["run_id"],
        replay=run_mode == LEDGER_MODE_REPLAY,
    )
    output_result = write_consensus_outputs(
        output_dir,
        batch=batch,
        reviews=reviews,
        summary_text=summary_text,
        run_records=run_records,
        funnel_records=funnel_records,
    )
    if run_mode == LEDGER_MODE_REPLAY:
        storage_result = {
            "status": "skipped",
            "write_performed": False,
            "reason": "replay_read_only",
            "run_id": batch["run_id"],
            "target_date": target_date,
        }
    else:
        try:
            storage_result = storage_service.append_daily_report_consensus(
                batch,
                reviews,
                funnel_records,
                file_hashes=output_result,
                allow_postgres=True,
            )
        except Exception as error:
            failed_output_dir = _quarantine_failed_consensus_outputs(output_result)
            raise ReportCriticalDataMissingError(
                "PostgreSQL 共识记录写入失败，停止生成；"
                f"本次五项产物已移至失败记录区：{failed_output_dir or '移动失败'}"
            ) from error
        if (
            not isinstance(storage_result, dict)
            or storage_result.get("write_performed") is not True
        ):
            failed_output_dir = _quarantine_failed_consensus_outputs(output_result)
            raise ReportCriticalDataMissingError(
                "PostgreSQL 共识记录未写入，停止生成：{}；"
                "本次五项产物已移至失败记录区：{}".format(
                    (
                        storage_result.get("reason")
                        if isinstance(storage_result, dict)
                        else "返回结果不正确"
                    )
                    or "原因未记录",
                    failed_output_dir or "移动失败",
                )
            )

    print(summary_text.rstrip())
    print(f"共识产物已生成: {output_result['output_dir']}")
    return {
        "batch": batch,
        "reviews": reviews,
        "consensus": consensus,
        "funnel_records": funnel_records,
        "decision_freeze": decision_freeze,
        "output": output_result,
        "storage": storage_result,
    }
