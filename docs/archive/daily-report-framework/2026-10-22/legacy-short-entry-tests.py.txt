class DailyReportConsensusEntryIntegrationTest(unittest.TestCase):
    def test_mode_resolution_keeps_fresh_default_full_until_calibration(self):
        import daily_report_entry_demo

        self.assertEqual(
            daily_report_entry_demo._resolve_daily_report_output_mode(
                SimpleNamespace(output_mode=None, replay=False)
            ),
            "full",
        )
        self.assertEqual(
            daily_report_entry_demo._resolve_daily_report_output_mode(
                SimpleNamespace(output_mode=None, replay=True),
                run_mode=daily_report_entry_demo.LEDGER_MODE_REPLAY,
            ),
            "full",
        )
        self.assertEqual(
            daily_report_entry_demo._resolve_daily_report_output_mode(
                SimpleNamespace(output_mode="short", replay=False)
            ),
            "short",
        )

    def test_no_report_and_output_mode_are_rejected_before_any_data_probe(self):
        import daily_report_entry_demo

        argv = [
            "daily_report_entry_demo.py",
            "2026-08-21",
            "--no-report",
            "--output-mode",
            "short",
        ]
        with patch.object(sys, "argv", argv), patch.object(
            daily_report_entry_demo,
            "_select_daily_report_data_path",
        ) as path_mock, patch.object(
            daily_report_entry_demo,
            "run_source_health_check",
        ) as health_mock, patch.object(
            daily_report_entry_demo,
            "run_reset_daily_price_runtime_query_cache",
            create=True,
        ) as reset_mock:
            with self.assertRaises(SystemExit) as error:
                daily_report_entry_demo.main()

        self.assertEqual(error.exception.code, 2)
        reset_mock.assert_not_called()
        path_mock.assert_not_called()
        health_mock.assert_not_called()

    def test_short_rejects_explicit_account_inputs_before_any_data_probe(self):
        import daily_report_entry_demo

        account_arguments = (
            ["--position", "我的仓位是甲公司 50%"],
            ["--tomorrow-empty"],
        )
        for extra_arguments in account_arguments:
            with self.subTest(arguments=extra_arguments):
                argv = [
                    "daily_report_entry_demo.py",
                    "2026-08-21",
                    "--output-mode",
                    "short",
                    *extra_arguments,
                ]
                with patch.object(sys, "argv", argv), patch.object(
                    daily_report_entry_demo,
                    "_select_daily_report_data_path",
                ) as path_mock, patch.object(
                    daily_report_entry_demo,
                    "run_source_health_check",
                ) as health_mock, patch.object(
                    daily_report_entry_demo,
                    "run_reset_daily_price_runtime_query_cache",
                    create=True,
                ) as reset_mock:
                    with self.assertRaises(SystemExit) as error:
                        daily_report_entry_demo.main()

                self.assertEqual(error.exception.code, 2)
                reset_mock.assert_not_called()
                path_mock.assert_not_called()
                health_mock.assert_not_called()

    def test_replay_short_writes_exactly_five_files_and_never_calls_database(self):
        import daily_report_entry_demo
        from report import runtime as report_runtime

        database_service = Mock()
        database_service.append_daily_report_consensus.side_effect = AssertionError(
            "replay must not call the database writer"
        )
        formal_decision_builder = Mock(
            wraps=daily_report_entry_demo.build_formal_account_decision
        )
        with TemporaryDirectory() as temporary, patch("builtins.print"), patch.object(
            daily_report_entry_demo,
            "build_formal_account_decision",
            formal_decision_builder,
        ), patch.object(
            report_runtime,
            "build_formal_account_decision",
            formal_decision_builder,
            create=True,
        ):
            result = daily_report_entry_demo._run_daily_report_consensus_short(
                _entry_consensus_data(),
                "2026-08-21",
                run_mode=daily_report_entry_demo.LEDGER_MODE_REPLAY,
                output_root=temporary,
                reviewer=_confirm_entry_proposal,
                model_version="fixture-model-v1",
                database_service=database_service,
                run_id="replay-short",
                now_fn=lambda: "2026-08-21T18:05:00+08:00",
            )
            formal_decision_builder.assert_called_once()

            output_dir = Path(result["output"]["output_dir"])
            self.assertEqual(
                {path.name for path in output_dir.iterdir()},
                set(OUTPUT_FILENAMES),
            )
            self.assertEqual(len(result["reviews"]), 10)
            self.assertEqual(result["consensus"]["ratio_label"], "10/10")
            self.assertEqual(result["batch"]["decision_hash"], (
                result["decision_freeze"]["decision_projection"]
                ["formal_account_decision"]["decision_hash"]
            ))
            self.assertEqual(
                result["batch"]["freeze_hash"],
                result["decision_freeze"]["freeze_hash"],
            )
            self.assertEqual(
                result["batch"]["decision_freeze"], result["decision_freeze"]
            )
            self.assertEqual(
                result["batch"]["decision_freeze"]["availability"], "complete"
            )
            projection = result["decision_freeze"]["decision_projection"]
            self.assertEqual(projection["availability"], "complete")
            self.assertEqual(projection["missing_fields"], [])
            for key in (
                "selection_workflow",
                "final_target_resolution",
                "position_decision",
                "two_layer_decision_payload",
                "formal_decision_input",
                "formal_account_decision",
            ):
                self.assertIsInstance(projection[key], dict)
                self.assertTrue(projection[key], key)
            self.assertFalse(result["storage"]["write_performed"])
            self.assertEqual(result["storage"]["reason"], "replay_read_only")
            self.assertEqual(list(Path(temporary).rglob("*.md")), [])
            self.assertEqual(list(Path(temporary).rglob("*.html")), [])

        database_service.append_daily_report_consensus.assert_not_called()

    def test_fresh_short_appends_batch_reviews_and_funnel_once(self):
        import daily_report_entry_demo

        database_service = Mock()
        database_service.is_postgres_backed.return_value = True
        database_service.append_daily_report_consensus.return_value = {
            "status": "inserted",
            "write_performed": True,
            "reason": "",
        }
        with TemporaryDirectory() as temporary, patch("builtins.print"):
            result = daily_report_entry_demo._run_daily_report_consensus_short(
                _entry_consensus_data(),
                "2026-08-21",
                run_mode=daily_report_entry_demo.LEDGER_MODE_FROZEN,
                output_root=temporary,
                reviewer=_confirm_entry_proposal,
                model_version="fixture-model-v1",
                database_service=database_service,
                run_id="fresh-short",
                now_fn=lambda: "2026-08-21T18:05:00+08:00",
            )

        database_service.append_daily_report_consensus.assert_called_once()
        call = database_service.append_daily_report_consensus.call_args
        self.assertEqual(len(call.args[1]), 10)
        self.assertEqual(call.args[0]["run_id"], "fresh-short")
        self.assertTrue(call.kwargs["allow_postgres"])
        self.assertEqual(set(call.kwargs["file_hashes"]["sha256"]), set(OUTPUT_FILENAMES))
        self.assertTrue(result["storage"]["write_performed"])

    def test_fresh_short_quarantines_files_when_database_write_does_not_complete(self):
        import daily_report_entry_demo
        from daily_report_render import ReportCriticalDataMissingError

        failure_modes = ("not_written", "exception")
        for failure_mode in failure_modes:
            with self.subTest(failure_mode=failure_mode), TemporaryDirectory() as temporary:
                database_service = Mock()
                database_service.is_postgres_backed.return_value = True
                if failure_mode == "not_written":
                    database_service.append_daily_report_consensus.return_value = {
                        "status": "conflict",
                        "write_performed": False,
                        "reason": "run_id_conflict",
                    }
                else:
                    database_service.append_daily_report_consensus.side_effect = RuntimeError(
                        "database unavailable"
                    )
                run_id = f"fresh-{failure_mode}"
                output_dir = consensus_output_directory(
                    temporary,
                    target_date="2026-08-21",
                    run_id=run_id,
                    replay=False,
                )

                with self.assertRaises(ReportCriticalDataMissingError):
                    daily_report_entry_demo._run_daily_report_consensus_short(
                        _entry_consensus_data(),
                        "2026-08-21",
                        run_mode=daily_report_entry_demo.LEDGER_MODE_FROZEN,
                        output_root=temporary,
                        reviewer=_confirm_entry_proposal,
                        model_version="fixture-model-v1",
                        database_service=database_service,
                        run_id=run_id,
                        now_fn=lambda: "2026-08-21T18:05:00+08:00",
                    )

                self.assertFalse(output_dir.exists())
                failed_dir = output_dir.parent / "_failed" / run_id
                self.assertEqual(
                    {path.name for path in failed_dir.iterdir()}, set(OUTPUT_FILENAMES)
                )

    def test_missing_formal_facts_stop_before_any_review_or_output(self):
        import daily_report_entry_demo
        from daily_report_render import ReportCriticalDataMissingError

        calls = []

        def reviewer(request):
            calls.append(request)
            return _confirm_entry_proposal(request)

        with TemporaryDirectory() as temporary:
            with self.assertRaises(ReportCriticalDataMissingError):
                daily_report_entry_demo._run_daily_report_consensus_short(
                    {"selection_workflow": _entry_consensus_data()["selection_workflow"]},
                    "2026-08-21",
                    run_mode=daily_report_entry_demo.LEDGER_MODE_REPLAY,
                    output_root=temporary,
                    reviewer=reviewer,
                    model_version="fixture-model-v1",
                )
            self.assertEqual(list(Path(temporary).iterdir()), [])
        self.assertEqual(calls, [])

    def test_main_replay_short_skips_full_report_bear_snapshot_and_old_sync(self):
        import daily_report_entry_demo

        class StubObserver:
            def __init__(self):
                self.steps = []
                self.finished = None

            def record_step(self, name, **kwargs):
                self.steps.append((name, kwargs))

            def finish(self, **kwargs):
                self.finished = kwargs

        observer = StubObserver()
        fake_paths = {
            filename: f"/tmp/replay-short/{filename}"
            for filename in OUTPUT_FILENAMES
        }
        short_result = {
            "output": {"paths": fake_paths},
            "consensus": {"ratio_label": "7/10"},
            "storage": {
                "write_performed": False,
                "reason": "replay_read_only",
            },
        }
        argv = [
            "daily_report_entry_demo.py",
            "2026-08-21",
            "--replay",
            "--output-mode",
            "short",
        ]
        data = {"position_context_provided": True, "target_date": "2026-08-21"}
        with patch.object(sys, "argv", argv), patch.object(
            daily_report_entry_demo,
            "run_reset_daily_price_runtime_query_cache",
            create=True,
        ), patch.object(
            daily_report_entry_demo,
            "_select_daily_report_data_path",
            return_value={
                "eligible": False,
                "selected_path": "replay",
                "reason": "显式复盘",
                "data": {},
            },
        ), patch.object(
            daily_report_entry_demo,
            "run_source_health_check",
            return_value={
                "overall_result": "ready",
                "stop_reasons": [],
                "warnings": [],
                "checks": [],
                "artifacts": {},
            },
        ), patch.object(
            daily_report_entry_demo,
            "render_source_health_terminal_lines",
            return_value=[],
        ), patch.object(
            daily_report_entry_demo,
            "replay_from_snapshot",
            return_value=(data, "2026-08-21"),
        ), patch.object(
            daily_report_entry_demo,
            "build_report_preflight",
            return_value={"failures": []},
        ), patch.object(
            daily_report_entry_demo,
            "render_report_preflight_lines",
            return_value=[],
        ), patch.object(
            daily_report_entry_demo,
            "run_current_report_blocks_new_entry",
            return_value=True,
            create=True,
        ), patch.object(
            daily_report_entry_demo,
            "_run_daily_report_consensus_short",
            return_value=short_result,
        ) as short_mock, patch.object(
            daily_report_entry_demo,
            "build_report_run_observer",
            return_value=observer,
        ), patch.object(
            daily_report_entry_demo,
            "build_report_quality_score",
            return_value={"schema_version": "report_quality_score.v1"},
        ), patch.object(
            daily_report_entry_demo,
            "build_report_outputs",
        ) as full_build_mock, patch.object(
            daily_report_entry_demo,
            "_render_daily_report_html_for_acceptance",
        ) as bear_mock, patch.object(
            daily_report_entry_demo,
            "save_report_outputs",
        ) as report_save_mock, patch.object(
            daily_report_entry_demo,
            "_save_data_snapshot_with_observation",
        ) as snapshot_mock, patch.object(
            daily_report_entry_demo,
            "import_daily_report_data_to_postgres",
        ) as old_sync_mock:
            result = daily_report_entry_demo.main()

        self.assertEqual(result, 0)
        short_mock.assert_called_once_with(
            data,
            "2026-08-21",
            run_mode=daily_report_entry_demo.LEDGER_MODE_REPLAY,
        )
        for skipped in (
            full_build_mock,
            bear_mock,
            report_save_mock,
            snapshot_mock,
            old_sync_mock,
        ):
            skipped.assert_not_called()
        self.assertEqual(observer.finished["output_payload"]["output_mode"], "short")
        self.assertFalse(observer.finished["output_payload"]["report_saved"])
        self.assertTrue(observer.finished["output_payload"]["consensus_saved"])
        self.assertEqual(
            set(observer.finished["output_payload"]["consensus_output_paths"]),
            set(OUTPUT_FILENAMES),
        )
